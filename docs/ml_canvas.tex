  \fontsize{7.2pt}{8.8pt}\selectfont
  \textbf{Core Objective:} Turn a single leaf photo into an immediate, actionable phytosanitary diagnosis within seconds, preventing catastrophic crop loss for underserved farmers.
  \vspace{1mm}
  \begin{itemize}
    \item \textbf{Global Problem Impact:} Crop diseases destroy \textbf{20--40\% global yield} annually (FAO), generating $>$\textbf{\$220B} in damages. Smallholders lack agronomists, and lab cultures take weeks, causing crop death and blind pesticide overuse.
    \item \textbf{Value Equation (Before $\rightarrow$ After):}
      \begin{itemize}[label={\textcolor{headValue}{$\rightarrow$}}]
        \item \emph{Turnaround:} Days/weeks (lab culture/PCR) $\rightarrow$ \textbf{$<5$ seconds} in-field.
        \item \emph{Cost per test:} \$10--50 (consultant/travel) $\rightarrow$ \textbf{$\sim$\$0.001} cloud inference.
        \item \emph{Efficacy:} Calendar chemical spraying $\rightarrow$ \textbf{Targeted, organic-first advisory}.
        \item \emph{Access:} 1 expert : 1,000 farmers $\rightarrow$ \textbf{1 model : millions via smartphone}.
      \end{itemize}
    \item \textbf{Target Personas \& Core Workflows:}
      \begin{itemize}
        \item \textbf{Smallholder Farmer} (0.5--5 ha, Kenya/India/Brazil): Photo $\rightarrow$ diagnosis + organic protocol in local language (\textbf{UC-01}, offline queue \textbf{UC-02}).
        \item \textbf{Extension Officer}: Batch upload 20+ farm photos $\rightarrow$ prioritized triage (\textbf{UC-03}).
        \item \textbf{Hobby / Urban Gardener}: Identifies benign spots $\rightarrow$ stops chemical waste.
        \item \textbf{Agri-Tech Platform / Co-op}: White-label diagnostic engine via REST/MCP (\textbf{UC-04}).
      \end{itemize}
    \item \textbf{System Interfaces:} Mobile PWA, WhatsApp/Telegram bot (MCP tool), IoT Gateway.
  \end{itemize}

  \vspace{1mm}
  \begin{itemize}
    \item \textbf{Reproducible Versioning \& Lineage (DVC):}
      \begin{itemize}
        \item Raw images and curated train/val/test splits tracked with Data Version Control (\textbf{DVC}). Immutable hash pointers stored in Git; raw assets stored in cloud object storage (S3/GCS).
        \item Pipelines locked via \texttt{dvc.yaml}; single-command reproduction via \texttt{dvc repro}.
      \end{itemize}
    \item \textbf{Active Learning \& Continuous Feedback Loop:}
      \begin{itemize}
        \item \textbf{Uncertainty Sampling:} Predictions with entropy in $[0.50, 0.75]$ automatically routed to human agronomist verification queue.
        \item \textbf{Dispute Ingestion:} Farmer feedback mechanism ("Dispute result" / "Report misdiagnosis") captures edge cases and seasonal disease variants.
      \end{itemize}
    \item \textbf{Collection Constraints \& Bottlenecks:}
      \begin{itemize}
        \item Expert labeling cost: Certified phytopathologist review $\sim$\$1.50/image.
        \item Class imbalance: Prevalent classes (Tomato Yellow Leaf Curl: 5,357 images) vs rare classes (Potato Early Blight: 1,000 images). Addressed via focal loss / reweighting.
      \end{itemize}
    \item \textbf{Privacy, Ethics \& Regulatory Boundaries:}
      \begin{itemize}
        \item \textbf{EXIF Stripping:} GPS coordinates discarded at ingestion; opt-in only for national disease tracking.
        \item \textbf{Zero PII:} Strictly non-human biological imagery.
        \item \textbf{Regulatory Disclaimer:} Certified as advisory support; not a legal pesticide mandate.

\begin{minipage}[t]{0.327\linewidth}%
  \begin{tcolorbox}[

    title={4. PREDICTION TASK (ML TASK)},
    boxed title style={colback=headModel},
    height=68mm
  ]
  \fontsize{7.2pt}{8.8pt}\selectfont
  \textbf{Mathematical Problem Formulation:}
  \vspace{1mm}
  \begin{itemize}
    \item \textbf{Task Archetype:} Supervised Multi-Class Single-Label Image Classification over $K = 38$ mutually exclusive botanical-condition classes.
    \item \textbf{Input Entity:} Single leaf RGB image tensor $\mathbf{x} \in \mathbb{R}^{3 \times 224 \times 224}$ under natural ambient field lighting and varying backgrounds.
    \item \textbf{Output Target:} Calibrated posterior class distribution $\mathbf{p} = \text{Softmax}(\mathbf{z}/\tau) \in [0,1]^{38}$ ($\sum_{k=1}^{38} p_k = 1$), where $\mathbf{z} \in \mathbb{R}^{38}$ represents raw logits and $\tau > 0$ is a post-hoc temperature scaling parameter calibrated on the validation set.
    \item \textbf{Delivered Inference Payload:}
      \begin{itemize}
        \item Top-$k$ ranked predictions: $\big(\text{Crop}_i, \text{Condition}_i, \hat{p}_i\big)$ for $i \in \{1,\dots, k\}$.
        \item Calibrated confidence score $C = \max_k p_k$.
        \item Out-of-Distribution (OOD) score $S_{\text{OOD}}(\mathbf{x})$ rejecting non-leaf/non-crop inputs.
        \item Human-readable agronomic advisory (pathogen type, severity, cultural remedy).
      \end{itemize}
    \item \textbf{Target Offline Acceptance Criteria:}
      \begin{itemize}
        \item Top-1 Accuracy: $\ge 95.0\%$ \quad\textbar\quad Macro-Averaged F1: $\ge 0.930$.
        \item Expected Calibration Error (ECE): $< 0.050$.
      \end{itemize}
  \end{itemize}
  \end{tcolorbox}%
\end{minipage}\hfill%
\begin{minipage}[t]{0.327\linewidth}%
  \begin{tcolorbox}[
    canvasbox,
    colback=bgFeatures,
    colframe=headFeatures,
    title={5. FEATURES \& PREPROCESSING},
    boxed title style={colback=headFeatures},
    height=68mm
  ]
  \fontsize{7.2pt}{8.8pt}\selectfont
  \textbf{Input Representations \& Data Transformations:}
  \vspace{1mm}
  \begin{itemize}
    \item \textbf{Deterministic Preprocessing Pipeline:}
      \begin{itemize}
        \item Format: 3-channel RGB image decode.
        \item Geometry: Aspect-ratio preserving resize and center-crop to $224 \times 224 \times 3$.
        \item Standardization: Zero-mean, unit-variance normalization with ImageNet statistics: $\boldsymbol{\mu} = [0.485, 0.456, 0.406]$, $\boldsymbol{\sigma} = [0.229, 0.224, 0.225]$.
      \end{itemize}
    \item \textbf{Robustness Augmentations (Invariance Training):}
      \begin{itemize}
        \item Spatial: Random rotation ($\pm 35^\circ$), horizontal/vertical flips, affine shearing, perspective transforms (simulating non-planar field angles).
        \item Photometric: Color jitter (brightness $\pm 0.2$, contrast $\pm 0.2$, saturation $\pm 0.2$, hue $\pm 0.05$), Gaussian blur (simulating smartphone defocus).
        \item Occlusion: Random Erasing / CutOut (simulating soil splatter or leaf tears).
      \end{itemize}
    \item \textbf{Deep Latent Representations:}
      \begin{itemize}
        \item High-dimensional feature vector $\mathbf{f} \in \mathbb{R}^{D}$ ($D=1280$ for EfficientNet, $D=768$ for ViT) extracted prior to the final classification layer.
      \end{itemize}
    \item \textbf{Inference Safeguards \& Guardrails:}
      \begin{itemize}
        \item Test-Time Augmentation (TTA, 5-crop averaging) on uncertain border cases.
        \item Energy-based OOD detector on $\mathbf{f}$ to reject non-botanical uploaded images.
      \end{itemize}
  \end{itemize}
  \end{tcolorbox}%
\end{minipage}\hfill%

    boxed title style={colback=headModel},

  ]
  \fontsize{7.2pt}{8.8pt}\selectfont
  \textbf{Model Architecture, MLOps Stack \& Orchestration:}
  \vspace{1mm}
  \begin{itemize}
    \item \textbf{Candidate Architectures (Transfer Learning):}
      \begin{itemize}
        \item \emph{Production Backbone:} \textbf{EfficientNet-B0} (4.0M parameters, compound scaling, lowest latency, ideal for CPU/edge container deployment).
        \item \emph{Robust Residual Baseline:} \textbf{ResNet-50 / ConvNeXt-Tiny} (robust feature extractor).
        \item \emph{Vision Transformer:} \textbf{ViT-Base/16} (global multi-head self-attention).
      \end{itemize}
    \item \textbf{Training Optimization Protocol:}
      \begin{itemize}
        \item Loss Function: Cross-Entropy with Label Smoothing ($\epsilon = 0.1$) to prevent overconfident misdiagnoses.
        \item Optimizer: AdamW ($\text{lr} = 3 \times 10^{-4}$, weight decay $10^{-4}$); Cosine Annealing schedule.
      \end{itemize}
    \item \textbf{Experiment Tracking \& Registry:}
      \begin{itemize}
        \item \textbf{MLflow}: Complete lineage tracking of hyperparameters, training curves, confusion matrices, and serialized model artifacts. Stage gates: \emph{Candidate} $\rightarrow$ \emph{Staging} $\rightarrow$ \emph{Production}.
      \end{itemize}
    \item \textbf{Pipeline Automation \& Quality Gates:}
      \begin{itemize}
        \item \textbf{Prefect DAGs}: Automated end-to-end retraining and evaluation workflows.
        \item \textbf{Great Expectations}: Strict data contract validation prior to training.
        \item \textbf{Deepchecks}: Automated model test suite verifying data leakage and performance.
        \item \textbf{CI/CD}: GitHub Actions running linter (\texttt{Ruff}), unit tests (\texttt{pytest}), container build.
      \end{itemize}
  \end{itemize}

  \vspace{1mm}
  \begin{itemize}
    \item \textbf{Confidence-Gated Operational Policy:}
      \begin{itemize}
        \item \textbf{High Confidence ($C \ge 0.80$):}
          \emph{Action:} Direct automated diagnosis. Render primary pathology name, severity rating, and step-by-step cultural/organic intervention (e.g., copper fungicide, canopy pruning).
        \item \textbf{Moderate Confidence ($0.60 \le C < 0.80$):}
          \emph{Action:} Top-2 differential diagnosis. Prompt user with a guided visual inspection checklist (e.g., \emph{"Look at leaf underside: are concentric rings or powdery spores visible?"}).
        \item \textbf{Low Confidence ($C < 0.60$) or OOD Flag:}
          \emph{Action:} \textbf{Strict Abstention Policy}. System outputs \emph{"Uncertain Diagnosis --- please consult a local extension specialist or re-photograph with clear lighting"}. Prevents dangerous pesticide misuse.
      \end{itemize}
    \item \textbf{Role-Specific Automated Decisions:}
      \begin{itemize}
        \item \emph{Farmer:} Instant plain-language advisory (English, Swahili, Hindi) delivered over WhatsApp or mobile app.
        \item \emph{Extension Officer:} Prioritized batch dashboard ranking farms by contagion severity for rapid field dispatch.
        \item \emph{Greenhouse IoT:} Automated activation of dehumidifiers / air scrubbers upon localized fungal detection.
      \end{itemize}

      Stratified $70\%$ Train / $15\%$ Validation / $15\%$ Test split, preserved across all 38 classes with fixed random seed and locked via DVC.
    \item \textbf{Target Offline Metrics:}
      \begin{itemize}
        \item Top-1 Accuracy: $\ge 95.0\%$ \quad\textbar\quad Macro-Averaged F1: $\ge 0.930$.
        \item Expected Calibration Error (ECE): $< 0.050$ (ensuring probabilities represent true frequencies).
      \end{itemize}
    \item \textbf{Asymmetric Cost-Utility Matrix (Impact Simulation):}
      \begin{itemize}
        \item \textbf{True Positive ($TP$):} \textcolor{headValue}{\textbf{High Positive Utility (+$100$)}} $\rightarrow$ Crop saved, contagion halted.
        \item \textbf{False Positive ($FP$)} (Healthy labeled as diseased): \textcolor{headDecision}{\textbf{Moderate Cost ($-15$)}} $\rightarrow$ Farmer spends \$5--15/ha on unnecessary organic spray; crop remains intact.
        \item \textbf{False Negative ($FN$)} (Diseased labeled as healthy): \textcolor{headEval}{\textbf{Catastrophic Loss ($-100$)}} $\rightarrow$ Unchecked pathogen proliferation, total crop destruction ($30$--$100\%$ yield collapse).
        \item \textbf{Pathogen Misidentification ($CM$):} \textcolor{headEval}{\textbf{Severe Loss ($-80$)}} $\rightarrow$ Ineffective treatment, phytotoxicity, delayed proper intervention.
      \end{itemize}
    \item \textbf{Mitigation:} Threshold tuning minimizing $FN$ for virulent classes + abstention fallback on uncertainty.
  \end{itemize}
